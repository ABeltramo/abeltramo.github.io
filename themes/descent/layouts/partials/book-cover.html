{{- /* Cover image for one book, or an empty string when there is none. Params: title, author, isbn, cover, sizes, class, eager. A file in assets/covers/.jpg wins. scripts/fetch-covers.mjs puts it there before the build, so the browser loads the cover from this site instead of Open Library. Without that file, the remote URL is used. */ -}} {{- $title := .title | default "" -}} {{- $author := .author | default "" -}} {{- $isbn := .isbn | default "" -}} {{- $cover := .cover | default "" -}} {{- $sizes := .sizes | default "96px" -}} {{- $class := .class | default "book-img" -}} {{- $eager := .eager | default false -}} {{- $alt := $title -}} {{- if $author }}{{ $alt = printf "%s by %s" $title $author }}{{ end -}} {{- if and $isbn (not $cover) -}} {{- with resources.Get (printf "covers/%s.jpg" $isbn) -}} {{- $small := .Resize "160x webp q80" -}} {{- $large := .Resize "320x webp q80" -}} {{- $out := printf `` $class $large.RelPermalink $small.RelPermalink $large.RelPermalink $sizes $large.Width $large.Height $alt (cond $eager "eager" "lazy") -}} {{- return ($out | safeHTML) -}} {{- end -}} {{- end -}} {{- with $cover -}} {{- $src := . -}} {{- if not (hasPrefix $src "http") }}{{ $src = $src | relURL }}{{ end -}} {{- $out := printf `` $class $src $alt (cond $eager "eager" "lazy") -}} {{- return ($out | safeHTML) -}} {{- else -}} {{- if $isbn -}} {{- $out := printf `` $class $isbn $alt (cond $eager "eager" "lazy") -}} {{- return ($out | safeHTML) -}} {{- end -}} {{- end -}} {{- return "" -}}
